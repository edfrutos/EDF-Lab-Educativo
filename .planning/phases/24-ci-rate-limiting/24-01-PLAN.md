  <action>
    1. `cd api && npm install express-rate-limit` (add to dependencies).
    2. In `auth.js`, add:
       ```javascript
       const rateLimit = require('express-rate-limit');

       function createLoginRateLimiter() {
         const windowMs = Number(process.env.LOGIN_RATE_LIMIT_WINDOW_MS) || 15 * 60 * 1000;
         const max = Number(process.env.LOGIN_RATE_LIMIT_MAX) || 10;
         return rateLimit({
           windowMs,
           max,
           standardHeaders: true,
           legacyHeaders: false,
           handler(req, res) {
             res.status(429).json({
               error: 'Demasiados intentos de inicio de sesión. Espera un momento e inténtalo de nuevo.'
             });
           }
         });
       }
       ```
    3. Export `createLoginRateLimiter` from `module.exports`.
  </action>
